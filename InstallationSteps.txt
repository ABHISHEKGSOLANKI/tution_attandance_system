## FRONTEND Commands
-------------------------
Step-1 
cd frontend
sudo nano /etc/nginx/sites-available/attendancesystem

server {

    listen 80;
    server_name 192.168.1.15;

    # =========================
    # FRONTEND
    # =========================
    location /attendancesystem/ {
        alias /var/www/attendancesystem/;
        index index.html;

        try_files $uri $uri/ /attendancesystem/index.html;
    }

    # =========================
    # PYTHON FACE API
    # =========================
    location /attendancesystem/api/faceapi/ {

        proxy_pass http://localhost:8098/;

        proxy_http_version 1.1;
		
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # =========================
    # SPRING BOOT BACKEND
    # =========================
    location /attendancesystem/api/ {

        proxy_pass http://localhost:8085/;

        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

Creating SSL self signed certificate
--------------------------------------------
sudo mkdir -p /etc/nginx/ssl
sudo openssl req -x509 -nodes -days 365 \
-newkey rsa:2048 \
-keyout /etc/nginx/ssl/local.key \
-out /etc/nginx/ssl/local.crt
